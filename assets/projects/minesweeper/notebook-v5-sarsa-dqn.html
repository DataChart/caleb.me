import random
import numpy as np
import gymnasium as gym
from gymnasium import spaces
import torch
import torch.nn as nn
import torch.optim as optim
import matplotlib.pyplot as plt

# Minesweeper Environment (Simplified)
class MinesweeperEnv(gym.Env):
    def __init__(self, width, height, n_mines,
                 rewards={'win': 50, 'lose': -10, 'progress': 1, 'guess': -0.3, 'no_progress': -0.1}):
        super().__init__()
        self.nrows, self.ncols = width, height
        self.ntiles = self.nrows * self.ncols
        self.n_mines = n_mines
        self.rewards = rewards
        self.action_space = spaces.Discrete(self.ntiles)
        self.observation_space = spaces.Box(low=-1, high=1, shape=(width, height, 1), dtype=np.float32)
        self.reset()

    def init_grid(self):
        board = np.zeros((self.nrows, self.ncols), dtype='object')
        mines = self.n_mines
        while mines > 0:
            row, col = random.randint(0, self.nrows-1), random.randint(0, self.ncols-1)
            if board[row][col] != 'B':
                board[row][col] = 'B'
                mines -= 1
        return board

    def get_neighbors(self, coord):
        x, y = coord
        neighbors = []
        for col in range(y-1, y+2):
            for row in range(x-1, x+2):
                if (x != row or y != col) and (0 <= row < self.nrows) and (0 <= col < self.ncols):
                    neighbors.append(self.grid[row, col])
        return np.array(neighbors)

    def count_bombs(self, coord):
        return np.sum(self.get_neighbors(coord) == 'B')

    def get_board(self):
        board = self.grid.copy()
        for x in range(self.nrows):
            for y in range(self.ncols):
                if board[x, y] != 'B':
                    board[x, y] = self.count_bombs((x, y))
        return board

    def get_state_im(self, state):
        state_im = [t['value'] for t in state]
        state_im = np.reshape(state_im, (self.nrows, self.ncols, 1)).astype(object)
        state_im[state_im == 'U'] = -1
        state_im[state_im == 'B'] = -2
        state_im = state_im.astype(np.int8) / 8
        return state_im.astype(np.float32)

    def init_state(self):
        unsolved_array = np.full((self.nrows, self.ncols), 'U', dtype='object')
        state = [{'coord': (x, y), 'value': 'U'} for x in range(self.nrows) for y in range(self.ncols)]
        return state, self.get_state_im(state)

    def click(self, action_index):
        coord = self.state[action_index]['coord']
        value = self.board[coord]
        if value == 'B' and self.n_clicks == 0:
            move = random.choice([i for i, s in enumerate(self.state) if self.board[s['coord']] != 'B'])
            coord = self.state[move]['coord']
            value = self.board[coord]
            self.state[move]['value'] = value
        else:
            self.state[action_index]['value'] = value
        if value == 0.0:
            self.reveal_neighbors(coord, clicked_tiles=[])
        self.n_clicks += 1

    def reveal_neighbors(self, coord, clicked_tiles):
        x, y = coord
        processed = clicked_tiles
        for col in range(y-1, y+2):
            for row in range(x-1, x+2):
                if (x != row or y != col) and (0 <= row < self.nrows) and (0 <= col < self.ncols):
                    if (row, col) not in processed:
                        processed.append((row, col))
                        index = next(i for i, t in enumerate(self.state) if t['coord'] == (row, col))
                        self.state[index]['value'] = self.board[row, col]
                        if self.board[row, col] == 0.0:
                            self.reveal_neighbors((row, col), clicked_tiles=processed)

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.n_clicks = 0
        self.n_progress = 0
        self.n_wins = 0
        self.grid = self.init_grid()
        self.board = self.get_board()
        self.state, self.state_im = self.init_state()
        return self.state_im, {}

    def step(self, action_index):
        done = False
        coords = self.state[action_index]['coord']
        current_state = self.state_im.copy()
        neighbors = self.get_neighbors(coords)
        self.click(action_index)
        new_state_im = self.get_state_im(self.state)
        self.state_im = new_state_im

        if self.state[action_index]['value'] == 'B':
            reward = self.rewards['lose']
            terminated = True
            truncated = False
        elif np.sum(new_state_im == -0.125) == self.n_mines:
            self.n_wins += 1
            reward = self.rewards['win']
            terminated = True
            truncated = False
        elif np.sum(new_state_im == -0.125) == np.sum(current_state == -0.125):
            reward = self.rewards['no_progress']
            terminated = False
            truncated = False
        elif all(t == -0.125 for t in neighbors):
            reward = self.rewards['guess']
            terminated = False
            truncated = False
        else:
            self.n_progress += 1
            reward = self.rewards['progress']
            terminated = False
            truncated = False

        info = {}
        return new_state_im, reward, terminated, truncated, info

# Q-Network
class QNetwork(nn.Module):
    def __init__(self, observation_space, action_space):
        super(QNetwork, self).__init__()
        self.flatten = nn.Flatten()
        self.fc1 = nn.Linear(observation_space.shape[0] * observation_space.shape[1] * observation_space.shape[2], 128)
        self.fc2 = nn.Linear(128, 128)
        self.fc3 = nn.Linear(128, action_space.n)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.flatten(x)
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        return self.fc3(x)

# DQN Agent
class DQNAgent:
    def __init__(self, observation_space, action_space, learning_rate=0.001, gamma=0.99, epsilon=1.0, epsilon_decay=0.0005, epsilon_min=0.01):
        self.q_network = QNetwork(observation_space, action_space)
        self.optimizer = optim.Adam(self.q_network.parameters(), lr=learning_rate)
        self.gamma = gamma
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min
        self.action_space = action_space

    def select_action(self, state):
        if random.random() < self.epsilon:
            return random.randrange(self.action_space.n)
        else:
            state = torch.from_numpy(state).float().unsqueeze(0)
            q_values = self.q_network(state)
            return torch.argmax(q_values).item()

    def update_epsilon(self):
        self.epsilon = max(self.epsilon_min, self.epsilon - self.epsilon_decay)

    def train_step(self, state, action, reward, next_state, done):
        state = torch.from_numpy(state).float().unsqueeze(0)
        next_state = torch.from_numpy(next_state).float().unsqueeze(0)
        reward = torch.tensor(reward).float().unsqueeze(0)
        done = torch.tensor(done).float().unsqueeze(0)

        q_values = self.q_network(state)
        next_q_values = self.q_network(next_state)

        q_value = q_values.gather(1, torch.tensor([[action]]))
        next_q_value = next_q_values.max(1)[0].unsqueeze(0)
        expected_q_value = reward + self.gamma * next_q_value * (1 - done)

        loss = (q_value - expected_q_value).pow(2).mean()

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        self.update_epsilon()

# SARSA Agent
class SARSAAgent:
    def __init__(self, observation_space, action_space, learning_rate=0.001, gamma=0.99, epsilon=1.0, epsilon_decay=0.0005, epsilon_min=0.01):
        self.q_network = QNetwork(observation_space, action_space)
        self.optimizer = optim.Adam(self.q_network.parameters(), lr=learning_rate)
        self.gamma = gamma
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min
        self.action_space = action_space

    def select_action(self, state):
        if random.random() < self.epsilon:
            return random.randrange(self.action_space.n)
        else:
            state = torch.from_numpy(state).float().unsqueeze(0)
            q_values = self.q_network(state)
            return torch.argmax(q_values).item()

    def update_epsilon(self):
        self.epsilon = max(self.epsilon_min, self.epsilon - self.epsilon_decay)

    def train_step(self, state, action, reward, next_state, next_action, done):
        state = torch.from_numpy(state).float().unsqueeze(0)
        next_state = torch.from_numpy(next_state).float().unsqueeze(0)
        reward = torch.tensor(reward).float().unsqueeze(0)
        done = torch.tensor(done).float().unsqueeze(0)

        q_values = self.q_network(state)
        next_q_values = self.q_network(next_state)

        q_value = q_values.gather(1, torch.tensor([[action]]))
        next_q_value = next_q_values.gather(1, torch.tensor([[next_action]])).squeeze(0)  # SARSA: Use Q-value of the *chosen* action
        expected_q_value = reward + self.gamma * next_q_value * (1 - done)

        loss = (q_value - expected_q_value).pow(2).mean()

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        self.update_epsilon()

# Training Loop for DQN
def train_dqn(env, agent, n_episodes=500):
    episode_rewards = []
    for episode in range(n_episodes):
        state, _ = env.reset()
        done = False
        total_reward = 0
        steps = 0

        while not done and steps < 200:
            action = agent.select_action(state)
            next_state, reward, terminated, truncated, info = env.step(action)
            done = terminated or truncated
            agent.train_step(state, action, reward, next_state, done)
            state = next_state
            total_reward += reward
            steps += 1

        episode_rewards.append(total_reward)

        if episode % 100 == 0:
            print(f"DQN Episode {episode}: Total Reward = {total_reward}, Epsilon = {agent.epsilon}")

    return episode_rewards

# Training Loop for SARSA
def train_sarsa(env, agent, n_episodes=500):
    episode_rewards = []
    for episode in range(n_episodes):
        state, _ = env.reset()
        done = False
        total_reward = 0
        steps = 0
        action = agent.select_action(state)  # Initial action

        while not done and steps < 200:
            next_state, reward, terminated, truncated, info = env.step(action)
            done = terminated or truncated
            next_action = agent.select_action(next_state)  # Choose next action *before* training step
            agent.train_step(state, action, reward, next_state, next_action, done)
            state = next_state
            action = next_action  # Update action for the next step
            total_reward += reward
            steps += 1

        episode_rewards.append(total_reward)

        if episode % 100 == 0:
            print(f"SARSA Episode {episode}: Total Reward = {total_reward}, Epsilon = {agent.epsilon}")

    return episode_rewards

# Visualization
def plot_rewards(rewards, title="Training - Total Reward per Episode"):
    plt.plot(rewards)
    plt.xlabel("Episode")
    plt.ylabel("Total Reward")
    plt.title(title)
    plt.grid(True)
    plt.show()
